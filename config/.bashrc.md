## conf

```bash
# --- Shortcuts for navigation and common commands ---
alias ..='cd ..'
alias c='clear'
alias a='ls -alh'
alias jj='vim .'
#alias fd='fdfind'
alias e='eza --long --icons --sort=type --tree --level=1'
alias ee='eza --long --icons --sort=type --tree --level=2'

# --- Shortcuts to edit config files ---
alias vc='vim ~/.vimrc'
alias bc='vim ~/.bashrc'
alias tc='vim ~/.tmux.conf'

# --- Tmux aliases ---
alias t='tmux'
alias ta='tmux attach'
alias tls='tmux ls'

# --- Ripgrep and fzf integration ---
# gf: search files by name with batcat preview
alias gf='fdfind --type f --hidden --follow --exclude .git | fzf --preview "batcat --style=numbers --color=always --line-range=:300 {}" --preview-window=right:60%'

# F: search file content with rg, batcat preview
alias F='rg --line-number --hidden --glob "!.git" --color=always . | fzf --ansi --delimiter=: --preview "batcat --style=numbers --color=always --highlight-line {2} {1}" --preview-window=right:60%:+{2}-/2'

# FF: same as F but ignores heavy dirs (node_modules, .venv, dist, build...)
alias FF='rg --line-number --hidden --color=always --glob "!.git" --glob "!node_modules" --glob "!.venv" --glob "!venv" --glob "!dist" --glob "!build" --glob "!.next" --glob "!__pycache__" --glob "!*.lock" . | fzf --ansi --delimiter=: --preview "batcat --style=numbers --color=always --highlight-line {2} {1}" --preview-window=right:60%:+{2}-/2'
```
